let currentUser = null;

function showMessage(text, type = "success") {
  document.getElementById("authMessage").innerHTML = `<div class="alert alert-\( {type}"> \){text}</div>`;
  setTimeout(() => document.getElementById("authMessage").innerHTML = "", 4000);

document.getElementById("showLogin").onclick = () => {

document.getElementById("registerBtn").onclick = async () => {
  const name = document.getElementById("regName").value.trim();
  const email = document.getElementById("regEmail").value.trim();
  const pass = document.getElementById("regPass").value;
  const ref = document.getElementById("regReferral").value.trim();
  if (!name || !email || !pass) return alert("Fill all fields");
  try {
    const cred = await createUserWithEmailAndPassword(auth, email, pass);
    const referralCode = name.substring(0, 4).toUpperCase() + Math.floor(Math.random() * 9000 + 1000);
    await setDoc(doc(db, "users", cred.user.uid), {
      name, email, balance: 0, referralCode, referredBy: ref || null, createdAt: serverTimestamp()
    });
    if (ref) {
      const q = query(collection(db, "users"), where("referralCode", "==", ref));
      const snap = await getDocs(q);
      if (!snap.empty) {
        const refUser = snap.docs[0];
        await updateDoc(refUser.ref, { balance: increment(200) });
        await addDoc(collection(db, "submissions"), {
          userId: refUser.id, userName: refUser.data().name, taskTitle: "Referral Bonus", reward: 200, status: "approved", date: new Date().toISOString()
        });
      }
    }
    showMessage("Account created! Please login.");
    document.getElementById("showLogin").click();
  } catch (e) { showMessage(e.message, "error"); }
};

document.getElementById("loginBtn").onclick = async () => {
  const email = document.getElementById("loginEmail").value.trim();
  const pass = document.getElementById("loginPass").value;
  try {
    if (email === ADMIN_EMAIL && pass === ADMIN_PASS) {
      currentUser = { isAdmin: true, email };
      sessionStorage.setItem("admin", "true");
      showAdmin();
      return;
    }
    const cred = await signInWithEmailAndPassword(auth, email, pass);
    const userDoc = await getDoc(doc(db, "users", cred.user.uid));
    currentUser = { uid: cred.user.uid, ...userDoc.data() };
    showUserDashboard();
  } catch (e) { showMessage(e.message, "error"); }
};

document.getElementById("logoutBtn").onclick = () => {
  signOut(auth);
  sessionStorage.removeItem("admin");
  location.reload();
};

onAuthStateChanged(auth, async (user) => {
  if (sessionStorage.getItem("admin") === "true") {
    showAdmin();
    return;
  }
  if (user) {
    const docSnap = await getDoc(doc(db, "users", user.uid));
    if (docSnap.exists()) {
      currentUser = { uid: user.uid, ...docSnap.data() };
      showUserDashboard();
    }
  }
});

async function showUserDashboard() {
  document.getElementById("authSection").classList.add("hidden");
  document.getElementById("userDashboard").classList.remove("hidden");
  document.getElementById("userInfo").classList.remove("hidden");
  document.getElementById("balanceDisplay").textContent = "₦" + (currentUser.balance || 0);
  document.getElementById("myReferralCode").value = currentUser.referralCode || "";
  loadTasks();
  loadMyHistory();
}

function showAdmin() {
  document.getElementById("authSection").classList.add("hidden");
  document.getElementById("adminPanel").classList.remove("hidden");
  document.getElementById("userInfo").classList.remove("hidden");
  document.getElementById("balanceDisplay").textContent = "ADMIN";
  loadAdmin();
}

async function loadTasks() {
  const snap = await getDocs(collection(db, "tasks"));
  const container = document.getElementById("tasksList");
  container.innerHTML = "";
  if (snap.empty) {
    await addDoc(collection(db, "tasks"), {
      title: "Watch video for 30 seconds",
      reward: 600,
      desc: "Watch the full video for 30 seconds to earn ₦600."
    });
    return loadTasks();
  }
  snap.forEach(docSnap => {
    const t = docSnap.data();
    const div = document.createElement("div");
    div.className = "task-card";
    div.innerHTML = `
      <h3>${t.title}</h3>
      <p>${t.desc}</p>
      <p><strong>Reward:</strong> ₦${t.reward}</p>
      <button class="success start-task" data-id="\( {docSnap.id}" data-title=" \){t.title}" data-reward="${t.reward}">Start Task</button>
    `;
    container.appendChild(div);
  });
  document.querySelectorAll(".start-task").forEach(btn => {
    btn.onclick = () => startTask(btn.dataset.id, btn.dataset.title, parseInt(btn.dataset.reward));
  });
}

let timerInterval;
function startTask(taskId, title, reward) {
  if (!confirm(`Start: \( {title}\nWait 30 seconds to claim ₦ \){reward}`)) return;
  let seconds = 30;
  const modal = document.createElement("div");
  modal.className = "card";
  modal.style.cssText = "position:fixed;top:50%;left:50%;transform:translate(-50%,-50%);z-index:1000;width:90%;max-width:400px;";
  modal.innerHTML = `
    <h3>${title}</h3>
    <div class="timer" id="timer">${seconds}s</div>
    <button id="claimBtn" class="success" disabled>Claim ₦${reward}</button>
    <button id="closeModal" class="secondary" style="margin-top:8px;">Cancel</button>
  `;
  document.body.appendChild(modal);
  const claimBtn = document.getElementById("claimBtn");
  timerInterval = setInterval(() => {
    seconds--;
    document.getElementById("timer").textContent = seconds + "s";
    if (seconds <= 0) {
      clearInterval(timerInterval);
      document.getElementById("timer").textContent = "Done!";
      claimBtn.disabled = false;
      claimBtn.onclick = async () => {
        await addDoc(collection(db, "submissions"), {
          userId: currentUser.uid,
          userName: currentUser.name,
          taskId, taskTitle: title, reward,
          status: "pending",
          date: new Date().toISOString()
        });
        alert("Submitted! Waiting for admin approval.");
        document.body.removeChild(modal);
        loadMyHistory();
      };
    }
  }, 1000);
  document.getElementById("closeModal").onclick = () => {
    clearInterval(timerInterval);
    document.body.removeChild(modal);
  };
}

async function loadMyHistory() {
  const q = query(collection(db, "submissions"), where("userId", "==", currentUser.uid));
  const snap = await getDocs(q);
  const el = document.getElementById("myHistory");
  if (snap.empty) { el.innerHTML = "<p>No earnings yet.</p>"; return; }
  let html = "<table><tr><th>Task</th><th>Reward</th><th>Status</th></tr>";
  snap.forEach(docSnap => {
    const s = docSnap.data();
    html += `<tr><td>\( {s.taskTitle}</td><td>₦ \){s.reward}</td><td>${s.status}</td></tr>`;
  });
  el.innerHTML = html + "</table>";
}

async function loadAdmin() {
  const usersSnap = await getDocs(collection(db, "users"));
  document.getElementById("statUsers").textContent = usersSnap.size;
  const pendingSnap = await getDocs(query(collection(db, "submissions"), where("status", "==", "pending")));
  document.getElementById("statPending").textContent = pendingSnap.size;
  const approvedSnap = await getDocs(query(collection(db, "submissions"), where("status", "==", "approved")));
  let total = 0;
  approvedSnap.forEach(d => total += d.data().reward);
  document.getElementById("statPaid").textContent = "₦" + total;

  const pendingDiv = document.getElementById("pendingList");
  pendingDiv.innerHTML = pendingSnap.empty ? "<p>No pending submissions</p>" : "";
  pendingSnap.forEach(docSnap => {
    const s = docSnap.data();
    const div = document.createElement("div");
    div.style.cssText = "margin:10px 0;padding:12px;border:1px solid #ddd;border-radius:8px;";
    div.innerHTML = `
      <strong>${s.userName}</strong> — \( {s.taskTitle} (₦ \){s.reward})<br>
      <button class="success approve-btn" style="width:auto;margin-top:8px;" data-id="\( {docSnap.id}" data-userid=" \){s.userId}" data-reward="${s.reward}">Approve</button>
      <button class="danger reject-btn" style="width:auto;margin-top:8px;" data-id="${docSnap.id}">Reject</button>
    `;
    pendingDiv.appendChild(div);
  });
  document.querySelectorAll(".approve-btn").forEach(btn => {
    btn.onclick = () => approve(btn.dataset.id, btn.dataset.userid, parseInt(btn.dataset.reward));
  });
  document.querySelectorAll(".reject-btn").forEach(btn => {
    btn.onclick = () => reject(btn.dataset.id);
  });

  const usersDiv = document.getElementById("usersList");
  usersDiv.innerHTML = "<table><tr><th>Name</th><th>Email</th><th>Balance</th><th>Referral</th></tr>";
  usersSnap.forEach(docSnap => {
    const u = docSnap.data();
    usersDiv.innerHTML += `<tr><td>\( {u.name}</td><td> \){u.email}</td><td>₦\( {u.balance}</td><td> \){u.referralCode}</td></tr>`;
  });
  usersDiv.innerHTML += "</table>";

  document.getElementById("addTaskBtn").onclick = async () => {
    const title = document.getElementById("newTaskTitle").value.trim();
    const reward = parseInt(document.getElementById("newTaskReward").value) || 0;
    const desc = document.getElementById("newTaskDesc").value.trim();
    if (!title) return alert("Enter title");
    await addDoc(collection(db, "tasks"), { title, reward, desc });
    document.getElementById("newTaskTitle").value = "";
    document.getElementById("newTaskDesc").value = "";
    alert("Task added!");
  };
}

async function approve(subId, userId, reward) {
  await updateDoc(doc(db, "submissions", subId), { status: "approved" });
  await updateDoc(doc(db, "users", userId), { balance: increment(reward) });
  loadAdmin();
}
async function reject(subId) {
  await updateDoc(doc(db, "submissions", subId), { status: "rejected" });
  loadAdmin();
}
